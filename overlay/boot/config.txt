# Raspberry Pi firmware config for omarchy-cm5 (CM5 / BCM2712).

# 16K-page Pi kernel from Arch Linux ARM. If a binary misbehaves on 16K
# pages, switch to the 4K build: kernel=kernel8.img via linux-rpi.
kernel=kernel8.img
arm_64bit=1

# Full KMS — required for Hyprland/wlroots-style compositors on v3d.
dtoverlay=vc4-kms-v3d-pi5
max_framebuffers=2
# 32bpp firmware framebuffer: the default 16-bit RGB565 simplefb makes
# plymouth render the splash as white garbage until vc4 KMS binds (~1s in).
framebuffer_depth=32
disable_overscan=1

# PCIe (NVMe on the carrier). Gen 3 is the usual CM5 carrier choice.
dtparam=pciex1
dtparam=pciex1_gen=3

initramfs initramfs-linux.img followkernel

# --- carrier-specific (TensorFleet) -----------------------------------------
# TODO: LVDS/DSI panel overlay + timings for vaio_cm5_carrier / koyomi-lvds-hat

# --- Hackberry Pi CM5 carrier (ZitaoTech) — validated 2026-09-13/14 --------
# Together with overlay/install/hardware/hackberry-cm5.sh this brings up:
# panel, battery gauge, onboard BT speakers, and the standalone Q20 keyboard.
dtoverlay=hackberrypicm5
dtparam=audio=on
dtoverlay=vc4-kms-dsi-pibrick

# Q20 standalone keyboard: the QMK/VIAL RP2040 controller (USB 4041:0001,
# "ZitaoTech HACKBerryPiQ20") is wired to the CM5 SoC's internal DWC2 USB2
# controller at /axi/usb@480000, which is disabled by default in the base DT.
# This mirrors the known-good old-image [cm5] section. dwc2 is built into
# the omega kernel (6.18.x-rpi-16k). The Q20 then appears as a normal USB HID
# keyboard + mouse (nub = BTN_LEFT; button-4 hold is BTN_RIGHT via QMK
# tap-dance; button-2 = Super; button-3 tap = Esc). No userspace remap helper
# is required.
[cm5]
dtoverlay=dwc2,dr_mode=host
